cmake_minimum_required(VERSION 3.16)
project(embodied_runtime C)
set(CMAKE_C_STANDARD 11)
add_library(er_runtime STATIC src/runtime.c)
target_include_directories(er_runtime PUBLIC include)
target_compile_options(er_runtime PRIVATE -Wall -Wextra -Werror)
add_library(er_ota STATIC src/ota.c)
target_include_directories(er_ota PUBLIC include)
target_compile_options(er_ota PRIVATE -Wall -Wextra -Werror)
add_executable(er_demo examples/demo.c)
target_link_libraries(er_demo PRIVATE er_runtime m)
enable_testing()
add_executable(er_tests tests/test_runtime.c)
target_link_libraries(er_tests PRIVATE er_runtime m)
add_test(NAME runtime COMMAND er_tests)
add_executable(er_ota_tests tests/test_ota.c)
target_link_libraries(er_ota_tests PRIVATE er_ota)
add_test(NAME ota COMMAND er_ota_tests)
