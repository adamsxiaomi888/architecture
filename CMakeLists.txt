cmake_minimum_required(VERSION 3.16)
project(embodied_runtime C)
set(CMAKE_C_STANDARD 11)
add_library(er_runtime STATIC src/runtime.c)
target_include_directories(er_runtime PUBLIC include)
target_compile_options(er_runtime PRIVATE -Wall -Wextra -Werror)
add_library(er_ota STATIC src/ota.c)
target_include_directories(er_ota PUBLIC include)
target_compile_options(er_ota PRIVATE -Wall -Wextra -Werror)
add_executable(er_demo examples/demo.c)
target_link_libraries(er_demo PRIVATE er_runtime m)
enable_testing()
add_executable(er_tests tests/test_runtime.c)
target_link_libraries(er_tests PRIVATE er_runtime m)
add_test(NAME runtime COMMAND er_tests)
add_executable(er_ota_tests tests/test_ota.c)
target_link_libraries(er_ota_tests PRIVATE er_ota)
add_test(NAME ota COMMAND er_ota_tests)
add_library(er_devices STATIC src/devices.c)
target_include_directories(er_devices PUBLIC include)
target_compile_options(er_devices PRIVATE -Wall -Wextra -Werror)
add_executable(er_device_tests tests/test_devices.c)
target_link_libraries(er_device_tests PRIVATE er_devices m)
add_test(NAME devices COMMAND er_device_tests)
add_library(er_sim STATIC sim/sim.c sim/actuator.c)
target_link_libraries(er_sim PUBLIC er_devices)
add_executable(er_sim_actuator_tests tests/test_sim_actuator.c)
target_link_libraries(er_sim_actuator_tests PRIVATE er_sim m)
add_test(NAME sim_actuator COMMAND er_sim_actuator_tests)
target_include_directories(er_sim PUBLIC include sim)
target_compile_options(er_sim PRIVATE -Wall -Wextra -Werror)
add_executable(er_sim_runner sim/runner.c)
target_link_libraries(er_sim_runner PRIVATE er_sim er_runtime m)
target_compile_options(er_sim_runner PRIVATE -Wall -Wextra -Werror)
foreach(scenario normal stale sensor stall)
    add_test(NAME sim_${scenario} COMMAND er_sim_runner ${scenario})
endforeach()
add_executable(er_web_tests tests/test_web_bridge.c web/wasm/bridge.c)
target_include_directories(er_web_tests PRIVATE web/wasm)
target_link_libraries(er_web_tests PRIVATE er_sim er_runtime m)
add_test(NAME web_bridge COMMAND er_web_tests)
find_program(ER_NODE_EXECUTABLE node)
if(ER_NODE_EXECUTABLE)
    add_test(NAME web_wasm COMMAND ${ER_NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_web_wasm.cjs)
endif()

add_library(er_osal STATIC src/osal.c)
target_include_directories(er_osal PUBLIC include)
target_compile_options(er_osal PRIVATE -Wall -Wextra -Werror)
add_executable(er_osal_tests tests/test_osal.c)
target_link_libraries(er_osal_tests PRIVATE er_osal)
add_test(NAME osal COMMAND er_osal_tests)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(er_osal_posix STATIC ports/posix/osal_posix.c)
    target_include_directories(er_osal_posix PUBLIC ports/posix)
    target_link_libraries(er_osal_posix PUBLIC er_osal rt)
    target_compile_options(er_osal_posix PRIVATE -Wall -Wextra -Werror)
    add_executable(er_periodic_demo examples/periodic.c)
    target_link_libraries(er_periodic_demo PRIVATE er_osal_posix)
    add_test(NAME posix_periodic COMMAND er_periodic_demo)
    set_tests_properties(posix_periodic PROPERTIES TIMEOUT 10)
    add_executable(er_control_posix examples/control_posix.c examples/control_task.c)
    target_include_directories(er_control_posix PRIVATE examples sim ports/posix)
    target_link_libraries(er_control_posix PRIVATE er_osal_posix er_devices er_sim er_runtime m)
    target_compile_options(er_control_posix PRIVATE -Wall -Wextra -Werror)
    add_test(NAME control_posix_normal COMMAND er_control_posix normal)
    add_test(NAME control_posix_stale COMMAND er_control_posix stale)
    add_test(NAME control_posix_sensor COMMAND er_control_posix sensor)
    set_tests_properties(control_posix_normal control_posix_stale control_posix_sensor PROPERTIES TIMEOUT 10)
endif()
